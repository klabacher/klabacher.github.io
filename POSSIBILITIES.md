## 🔄 Redux Toolkit

Centralized state management with Redux Toolkit.

**Location:** `src/store/`

**Store Structure:**
```
src/store/
├── store.ts              # Store configuration
└── slices/
    └── appSlice.ts       # App state slice (theme, etc.)
```

**Usage:**
```tsx
import { useSelector, useDispatch } from 'react-redux';
import { selectTheme, toggleTheme } from './store/slices/appSlice';

function ThemeToggle() {
  const theme = useSelector(selectTheme);
  const dispatch = useDispatch();

  return (
    <button onClick={() => dispatch(toggleTheme())}>
      Current: {theme}
    </button>
  );
}
```

**Adding a new slice:**
```tsx
// src/store/slices/userSlice.ts
import { createSlice, PayloadAction } from '@reduxjs/toolkit';

interface UserState {
  name: string | null;
  isAuthenticated: boolean;
}

const initialState: UserState = {
  name: null,
  isAuthenticated: false,
};

export const userSlice = createSlice({
  name: 'user',
  initialState,
  reducers: {
    login: (state, action: PayloadAction<string>) => {
      state.name = action.payload;
      state.isAuthenticated = true;
    },
    logout: (state) => {
      state.name = null;
      state.isAuthenticated = false;
    },
  },
});

export const { login, logout } = userSlice.actions;
export default userSlice.reducer;
```

Declarative routing for React applications.

**Location:** `src/main.tsx`, `src/App.tsx`

**Current Routes:**

- `/about` - About page

**Usage:**
```tsx

navigate('/about');

```

## 🌐 Internationalization (i18n)

Multi-language support with react-i18next.

**Location:** `src/i18n/`

**Structure:**
```
src/i18n/
├── config.ts              # i18n configuration
└── locales/
    ├── en.json            # English translations
    ├── pt.json            # Portuguese translations
    └── es.json            # Spanish translations
```

**Usage:**
```tsx
import { useTranslation } from 'react-i18next';

function MyComponent() {
  const { t, i18n } = useTranslation();

  return (
    <div>
      <h1>{t('hero.title')}</h1>
      <p>{t('hero.subtitle')}</p>

      {/* Change language */}
      <button onClick={() => i18n.changeLanguage('pt')}>
        Português
      </button>
    </div>
  );
}
```

**Adding translations:**
```json
// src/i18n/locales/en.json
{
  "myComponent": {
    "title": "Welcome",
    "greeting": "Hello, !"
  }
}
```

```tsx
// Usage with interpolation
t('myComponent.greeting', { name: 'John' })
// Output: "Hello, John!"
```

## 🧪 Testing

Unit and component testing with Vitest and Testing Library.

**Location:** `src/__tests__/`, `vitest.config.ts`

**Commands:**
```bash
 run test           # Run tests once
 run test:watch     # Watch mode
 run test:ui        # Vitest UI
 run test:coverage  # With coverage report
```

**Writing tests:**
```tsx
// src/__tests__/MyComponent.test.tsx
import { describe, it, expect } from 'vitest';
import { render, screen } from '@testing-library/react';
import userEvent from '@testing-library/user-event';
import MyComponent from '../MyComponent';

describe('MyComponent', () => {
  it('renders correctly', () => {
    render(<MyComponent />);
    expect(screen.getByText('Hello')).toBeInTheDocument();
  });

  it('handles click', async () => {
    const user = userEvent.setup();
    render(<MyComponent />);

    await user.click(screen.getByRole('button'));
    expect(screen.getByText('Clicked!')).toBeInTheDocument();
  });
});
```

**Test utilities:** See `src/test/test-utils.tsx` for custom render with providers.
